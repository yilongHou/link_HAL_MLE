import numpy as np
import pandas as pd
from scipy.stats import truncnorm, uniform
from rpy2.robjects import pandas2ri, r

# Helper functions
def generate_truncated_normal(n, mean=0.5, std=0.1, lower=0, upper=1):
    a, b = (lower - mean)/std, (upper - mean)/std
    return truncnorm.rvs(a, b, loc=mean, scale=std, size=n)

def generate_uniform_censor(n, lower=0, upper=1):
    return uniform.rvs(lower, upper - lower, size=n)

def convert_to_survival_data(T, C):
    T_tilde = np.minimum(T, C)
    delta = (T <= C).astype(int)
    return pd.DataFrame({"T_tilde": T_tilde, "delta": delta})

def true_density(grid, mean=0.5, std=0.1, lower=0, upper=1):
    a, b = (lower - mean)/std, (upper - mean)/std
    return truncnorm.pdf(grid, a, b, loc=mean, scale=std)

def kl_divergence(p, q, grid):
    dx = grid[1] - grid[0]
    mask = (p > 0) & (q > 0)
    return np.sum(p[mask] * np.log(p[mask] / q[mask])) * dx

def simulate_kl(seed):
    np.random.seed(seed)
    n = 1000
    T = generate_truncated_normal(n)
    C = generate_uniform_censor(n)
    data = convert_to_survival_data(T, C)

    observed = data.loc[data.delta==1, "T_tilde"]
    censored = data.loc[data.delta==0, "T_tilde"]

    # Convert to R objects
    T_r      = pandas2ri.py2rpy(pd.Series(T))
    obs_r    = pandas2ri.py2rpy(observed)
    cen_r    = pandas2ri.py2rpy(censored)
    grid_pts = np.linspace(0, 1, 1000)
    grid_r   = pandas2ri.py2rpy(pd.Series(grid_pts))

    # Fit censored with oldlogspline
    fit_cens = logspline.oldlogspline(
        uncensored=obs_r,
        right=cen_r,
        lbound=0,
        ubound=1
    )
    dens_cens = np.array(r["doldlogspline"](grid_r, fit_cens))

    # Fit full data with logspline
    fit_full = logspline.logspline(x=T_r)
    dens_full = np.array(r["dlogspline"](grid_r, fit_full))

    dens_true = true_density(grid_pts)
    kl_cens = kl_divergence(dens_true, dens_cens, grid_pts)
    kl_full = kl_divergence(dens_true, dens_full, grid_pts)
    return kl_cens, kl_full

# Run for 500 simulations with progress bar
base = 12776
results = []
for i in tqdm(range(1, 501), desc="Simulations"):
    seed = base + i
    kl_c, kl_f = simulate_kl(seed)
    results.append({"iteration": i, "seed": seed, "kl_censored": kl_c, "kl_full": kl_f})

df = pd.DataFrame(results)

    return uniform.rvs(lower, upper - lower, size=n)

def convert_to_survival_data(T, C):
    """Return DataFrame with (T̃, δ)."""
    T_tilde = np.minimum(T, C)
    delta   = (T <= C).astype(int)
    return pd.DataFrame({"T_tilde": T_tilde, "delta": delta})

def true_density(grid, mean=0.5, std=0.1, lower=0, upper=1):
    a, b = (lower - mean) / std, (upper - mean) / std
    return truncnorm.pdf(grid, a, b, loc=mean, scale=std)

def fit_logspline_pair(seed, n=1000, grid=np.linspace(0, 1, 1000)):
    """
    Simulate one data set (with censoring) under given seed
    and return densities from:
      • oldlogspline   (censored data only)
      • logspline      (full, uncensored data)
    """
    np.random.seed(seed)

    # --- simulate data -------------------------------------------------
    T = generate_truncated_normal(n)
    C = generate_uniform_censor(n)
    data = convert_to_survival_data(T, C)

    # split observed and censored event times for oldlogspline
    obs_r  = pandas2ri.py2rpy(data.loc[data.delta == 1, "T_tilde"])
    cen_r  = pandas2ri.py2rpy(data.loc[data.delta == 0, "T_tilde"])
    T_r    = pandas2ri.py2rpy(pd.Series(T))                    # full data
    grid_r = pandas2ri.py2rpy(pd.Series(grid))

    # --- fit censored model --------------------------------------------
    fit_cens  = logspline.oldlogspline(uncensored=obs_r, right=cen_r,
                                       lbound=0, ubound=1)
    dens_cens = np.array(r["doldlogspline"](grid_r, fit_cens))

    # --- fit full‑data model ------------------------------------------
    fit_full  = logspline.logspline(x=T_r)      # default density fit
    dens_full = np.array(r["dlogspline"](grid_r, fit_full))

    return dens_cens, dens_full
